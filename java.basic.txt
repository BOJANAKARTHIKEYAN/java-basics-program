import java.util.Scanner;

public class ReverseNumber {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int rev = 0;

        if (n != 0) {
            while (n != 0) {
                rev = rev * 10 + n % 10;
                n = n / 10;
            }
        } else {
            rev = 0;
        }

        System.out.println(rev);
        sc.close();
    }
}

import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int original = n;
        int sum = 0;

        while (n != 0) {
            int digit = n % 10;
            sum += digit * digit * digit; // cube of each digit
            n = n / 10;
        }

        if (sum == original) {
            System.out.println(original);
        } else {
            System.out.println("Not an Armstrong number");
        }

        sc.close();
    }
}

import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int ans =n;
        int digit =0;
        int res=0;
        while(n>0){
            n=n/10;
            digit++;
        }
        n= ans;
        while(n>0){
            int r=n%10;
            int mul=1;
            for(int i=1;i<=digit;i++){
                mul=mul*r;
            }
            res = res+mul;
            n=n/10;
 
        }
        System.out.println((res==ans)?"amstrong":"not");
    }
}

import java.util.*;
public class Main
{
 	public static void main(String[] args) {
 		Scanner sc = new Scanner(System.in);
 		int size = sc.nextInt();
 		int arr[] = new int [size];
 		int oddsum=0,evensum=0;
 		for(int i=0;i<size;i++){
 		    arr[i]=sc.nextInt();
 		    if(arr[i]%2==0){
 		        evensum=evensum+arr[i];
 		    }else{
 		        oddsum = oddsum+arr[i];
 		    }
 
 		}
 		 System.out.print(evensum>oddsum?evensum-oddsum:oddsum+evensum);
 	}
}
